Sample PostGIS Query Production 
SELECT
  service_name,
  address,
  ST_Distance(
    geom,
    ST_SetSRID(ST_MakePoint($lon, $lat), 4326)::geography
  ) as distance_meters
FROM "ServiceRequest"
WHERE ST_DWithin(
  geom,
  ST_SetSRID(ST_MakePoint($lon, $lat), 4326)::geography,
  500  -- radius in meters
)
ORDER BY distance_meters
LIMIT 100;
